import java.util.*;
public class ham
{
    boolean prime(int n)
    {
        int c=0;
        for(int i=2;i<n/2;i++)
        {
            if(n%i==0)
            c++;
        }
        if(c==0)
        return true;
        else
        return false;
    }
    void main()
    {   
        Scanner sc=new Scanner(System.in);
        System.out.println("Enter the number:");
        int n=sc.nextInt();int flag=0;
        for(int i=2;i<=n/2;i++)
        {
            if(n%i==0&&prime(i))
            {
                System.out.print(i+" ");
                if(i==2||i==3||i==5)
                flag=1;
                else
                {
                    flag=0;break;
                }
            }
        }
        System.out.println();
        if(flag==1)
        {
            System.out.println(n+" is a Hamming number");
        }
        else
        System.out.println(n+" is not a Hamming number");
    }
}
